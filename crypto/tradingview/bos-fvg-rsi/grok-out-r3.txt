    float buf = stopBufPct / 100.0
    stopMode == "last_low" ? l2 * (1.0 - buf) : stopMode == "zone" ? zoneBot * (1.0 - buf) : entryPx - atrMult * atrNow

// [R9] short stop (mirror)
f_shortStop(float entryPx, float h2, float zoneTop, float atrNow) =>
    float buf = stopBufPct / 100.0
    stopMode == "last_low" ? h2 * (1.0 + buf) : stopMode == "zone" ? zoneTop * (1.0 + buf) : entryPx + atrMult * atrNow
