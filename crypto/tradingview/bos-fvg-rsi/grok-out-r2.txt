// RULE → CODE MAP
//   [R1]  Pivots (ta.pivothigh / ta.pivotlow, pivotLen) .............. Structure
//   [R2]  Bearish context L1→L2 lower-low + H_int (green line) ...... Structure
//   [R3]  RSI at L2 bar (rsi[pivotLen] on confirm) < rsiThresh ...... RSI Filter
//   [R4]  BoS = close > H_int; fail = close < L2; armTimeout ........ Structure
//   [R5]  impulseHigh tracked from BoS until entry/cancel ........... Structure
//   [R6]  Bullish FVG in leg L2..BoS+3; most recent; zone high[2]–low Entry
//   [R7]  Limit at FVG top|mid; entryTimeout; miss if TP1 run; skip . Entry
//   [R8]  One position; pyramiding=0 ................................ Declaration
//   [R9]  Stop modes: last_low | zone | atr ......................... Exits
//   [R10] TP1 at impulseHigh, qty_percent=tp1Pct; optional BE ....... Exits
//   [R11] Runner: supertrend flip | MA close; hard stop always ...... Exits
//   [R12] Shorts: full mirror when enableShorts ..................... Shorts
//   [R13] No repaint: no request.security, no lookahead; pivots late Declaration
//   [R14] Commission %, slippage ticks, % equity sizing ............. Costs
//   [R15] Date-range filter gates entries ........................... Costs & Dates
//   [R16] Visuals: H_int line, FVG box, SL/TP lines, markers ........ Visuals
//
// NOTE: Commission/slippage/position size: adjust in Settings → Properties panel.
//
// KNOWN LIMITATIONS
//   • Pivot confirmation lags by pivotLen bars (accepted; never look-ahead).
//   • Single position only (pyramiding=0); opposing setups ignored while busy.
//   • Setups with no FVG in the L2→BoS+3 leg are skipped and counted
//     (skippedNoFvgLong / skippedNoFvgShort); no market chase.
//   • Strategy engine fills limits intrabar; TP1/BE detection is bar-based.
//   • Missed-trade cancel: after BoS, close through running impulse extreme while
//     low/high never trades the limit (retrace never touches entry).
//   • On the fill bar itself the protective stop is not yet active intrabar
//     (exits are placed at end of the fill bar's calculation) — standard
//     bar-based backtest caveat.
// =============================================================================

// ─── Pure helpers ────────────────────────────────────────────────────────────
// [R9] long stop
f_longStop(float entryPx, float l2, float fvgBot, float atrNow) =>
    float buf = stopBufPct / 100.0
    stopMode == "last_low" ? l2 * (1.0 - buf) : stopMode == "zone" ? fvgBot * (1.0 - buf) : entryPx - atrMult * atrNow

// [R9] short stop (mirror)
f_shortStop(float entryPx, float h2, float fvgTop, float atrNow) =>
    float buf = stopBufPct / 100.0
    stopMode == "last_low" ? h2 * (1.0 + buf) : stopMode == "zone" ? fvgTop * (1.0 + buf) : entryPx + atrMult * atrNow

// [R7] entry from FVG — long near-edge = top; short near-edge = bottom
f_longEntry(float top, float bot) =>
    entryMode == "FVG top" ? top : (top + bot) / 2.0

f_shortEntry(float top, float bot) =>
    entryMode == "FVG top" ? bot : (top + bot) / 2.0

// Scan most-recent bullish FVG in [leftBar, rightBar] ending at current bar
f_scanBullFvg(int leftBar, int rightBar) =>
    float bestTop = na
    float bestBot = na
    int   bestBar = na
    bool  found   = false
    int   back    = math.max(bar_index - leftBar, 0)
    if back >= 2
        for i = back to 0
            int idx = bar_index - i
            if idx >= leftBar and idx <= rightBar
                if low[i] > high[i + 2]
                    bestTop := low[i]
                    bestBot := high[i + 2]
                    bestBar := idx
                    found   := true
    [found, bestTop, bestBot, bestBar]

// Scan most-recent bearish FVG in [leftBar, rightBar]
f_scanBearFvg(int leftBar, int rightBar) =>
    float bestTop = na
    float bestBot = na
    int   bestBar = na
    bool  found   = false
    int   back    = math.max(bar_index - leftBar, 0)
    if back >= 2
        for i = back to 0
            int idx = bar_index - i
            if idx >= leftBar and idx <= rightBar
                if high[i] < low[i + 2]
                    bestTop := low[i + 2]
                    bestBot := high[i]
                    bestBar := idx
                    found   := true
    [found, bestTop, bestBot, bestBar]

// ─── Busy / flat gates [R8] ──────────────────────────────────────────────────
bool flat      = strategy.position_size == 0
bool longBusy  = longState == S_ENTRY or longState == S_TRADE or strategy.position_size > 0
bool shortBusy = shortState == S_ENTRY or shortState == S_TRADE or strategy.position_size < 0
bool canArmLong  = flat and not longBusy and shortState != S_ENTRY and shortState != S_TRADE and longState != S_ENTRY and longState != S_TRADE
bool canArmShort = flat and not shortBusy and longState != S_ENTRY and longState != S_TRADE and shortState != S_ENTRY and shortState != S_TRADE

// =============================================================================
// [R1][R2][R3] Pivot chain — LONG lower-low + interim high
// =============================================================================
if not na(ph)
    int phBar = bar_index - pivotLen
    // Highest confirmed PH after current L1 candidate
    if not na(trkPL) and phBar > trkPLBar
        if na(trkPH) or ph >= trkPH
            trkPH    := ph
            trkPHBar := phBar

if not na(pl)
    int plBar = bar_index - pivotLen
    // [R2] L1 then L2 < L1 with H_int between
    if not na(trkPL) and pl < trkPL and not na(trkPH) and trkPHBar > trkPLBar and trkPHBar < plBar
        // [R3] RSI on actual L2 pivot bar via history offset at confirmation
        float rsiAtL2 = rsi[pivotLen]
        bool rsiOk = not na(rsiAtL2) and rsiAtL2 < rsiThresh
        if rsiOk and inDateRange and canArmLong
            // Cancel prior arm visuals if re-arming
            if longState == S_ARMED
                if not na(lnLongHint)
                    line.delete(lnLongHint)
                    lnLongHint := na
            longState       := S_ARMED
            longL1          := trkPL
            longL2          := pl
            longHint        := trkPH
            longL1Bar       := trkPLBar
            longL2Bar       := plBar
            longHintBar     := trkPHBar
            longArmedBar    := bar_index
            longBosBar      := na
            longImpulseHigh := na
            longFvgFound    := false
            longFvgTop      := na
            longFvgBot      := na
            longFvgBar      := na
            longTp1Done     := false
            markLongL2      := true
            // [R16] green line = H_int
            lnLongHint := line.new(longHintBar, longHint, bar_index, longHint, color = color.new(color.green, 0), width = 2)
            if showRsiLbl
                if not na(lbLongRsi)
                    label.delete(lbLongRsi)
                lbLongRsi := label.new(plBar, pl, "L2 RSI " + str.tostring(rsiAtL2, "#.##"), style = label.style_label_up, color = color.new(color.green, 70), textcolor = color.green, size = size.tiny)
    // Advance long low chain
    trkPL    := pl
    trkPLBar := plBar
    trkPH    := na
    trkPHBar := na

// =============================================================================
// [R12] Pivot chain — SHORT higher-high + interim low
// =============================================================================
if enableShorts and not na(pl)
    int plBarS = bar_index - pivotLen
    if not na(trkPH_s) and plBarS > trkPHBar_s
        if na(trkPL_s) or pl <= trkPL_s
            trkPL_s    := pl
            trkPLBar_s := plBarS

if enableShorts and not na(ph)
    int phBarS = bar_index - pivotLen
    if not na(trkPH_s) and not na(trkPL_s) and ph > trkPH_s and trkPLBar_s > trkPHBar_s and trkPLBar_s < phBarS
        // [R12] RSI at H2 bar > 100 - rsiThresh
        float rsiAtH2 = rsi[pivotLen]
        bool rsiOkS = not na(rsiAtH2) and rsiAtH2 > (100.0 - rsiThresh)
        if rsiOkS and inDateRange and canArmShort
            if shortState == S_ARMED
                if not na(lnShortLint)
                    line.delete(lnShortLint)
                    lnShortLint := na
            shortState       := S_ARMED
            shortH1          := trkPH_s
            shortH2          := ph
            shortLint        := trkPL_s
            shortH1Bar       := trkPHBar_s
            shortH2Bar       := phBarS
            shortLintBar     := trkPLBar_s
            shortArmedBar    := bar_index
            shortBosBar      := na
            shortImpulseLow  := na
            shortFvgFound    := false
            shortFvgTop      := na
            shortFvgBot      := na
            shortFvgBar      := na
            shortTp1Done     := false
            markShortH2      := true
            // [R16] red line = L_int
            lnShortLint := line.new(shortLintBar, shortLint, bar_index, shortLint, color = color.new(color.red, 0), width = 2)
            if showRsiLbl
                if not na(lbShortRsi)
                    label.delete(lbShortRsi)
                lbShortRsi := label.new(phBarS, ph, "H2 RSI " + str.tostring(rsiAtH2, "#.##"), style = label.style_label_down, color = color.new(color.red, 70), textcolor = color.red, size = size.tiny)
        // Next chain starts at this high
        trkPH_s    := ph
        trkPHBar_s := phBarS
        trkPL_s    := na
        trkPLBar_s := na
    else if na(trkPH_s) or na(trkPL_s)
        // Seed / refresh H1 while waiting for interim low
        trkPH_s    := ph
        trkPHBar_s := phBarS
        if not na(trkPL_s) and phBarS > trkPLBar_s
            // PH after L but not higher-high → restart
            trkPL_s    := na
            trkPLBar_s := na
    else
        // Have H1+L but this PH is not a higher high → restart from PH
        trkPH_s    := ph
        trkPHBar_s := phBarS
        trkPL_s    := na
        trkPLBar_s := na

// =============================================================================
// LONG state machine [R4][R5][R6][R7][R9][R10][R11]
// =============================================================================
// [R16] extend H_int while armed/entry
if (longState == S_ARMED or longState == S_ENTRY) and not na(lnLongHint) and not na(longHint)
    line.set_x2(lnLongHint, bar_index)
    line.set_y1(lnLongHint, longHint)
    line.set_y2(lnLongHint, longHint)

// [R4] ARMED
if longState == S_ARMED
    if close < longL2
        // structure failed
        strategy.cancel("Long")
        if not na(lnLongHint)
            line.delete(lnLongHint)
            lnLongHint := na
        longState := S_IDLE
    else if bar_index - longArmedBar >= armTimeout
        strategy.cancel("Long")
        if not na(lnLongHint)
            line.delete(lnLongHint)
            lnLongHint := na
        longState := S_IDLE
    else if close > longHint
        // BoS
        longState       := S_ENTRY
        longBosBar      := bar_index
        longImpulseHigh := high
        markLongBos     := true
        // [R6] initial FVG scan L2 .. BoS (+ room to BoS+3 later)
        [fnd, t, b, fb] = f_scanBullFvg(longL2Bar, longBosBar + 3)
        longFvgFound := fnd
        longFvgTop   := t
        longFvgBot   := b
        longFvgBar   := fb
        if not na(lnLongImp)
            line.delete(lnLongImp)
        lnLongImp := line.new(longBosBar, longImpulseHigh, bar_index, longImpulseHigh, color = color.new(color.teal, 0), width = 1, style = line.style_dashed)

// [R5][R6][R7] ENTRY
if longState == S_ENTRY
    // [F4] capture prior impulse extreme before this bar's update
    float prevImp = longImpulseHigh
    longImpulseHigh := math.max(nz(longImpulseHigh, high), high)
    if not na(lnLongImp)
        line.set_x2(lnLongImp, bar_index)
        line.set_y1(lnLongImp, longImpulseHigh)
        line.set_y2(lnLongImp, longImpulseHigh)

    // Live FVG updates through BoS+3
    if bar_index >= longL2Bar and bar_index <= longBosBar + 3
        if low > high[2]
            longFvgTop   := low
            longFvgBot   := high[2]
            longFvgBar   := bar_index
            longFvgFound := true

    // [F2] FIRST: fill check — never cancel while holding a filled long
    if strategy.position_size > 0 and strategy.position_size[1] <= 0
        longState   := S_TRADE
        longPosMax  := strategy.position_size
        longTp1Done := false
        longEntryPx := strategy.position_avg_price
        longTp1Px   := longImpulseHigh
        longStopPx  := f_longStop(longEntryPx, longL2, longFvgBot, atr)
        // [R9][R10][F3] TP1 tranche + runner always stop-protected
        if tp1Pct > 0
            strategy.exit("Long-TP1", from_entry = "Long", qty_percent = tp1Pct, limit = longTp1Px, stop = longStopPx, comment_profit = "TP1", comment_loss = "SL")
            strategy.exit("Long-RUN", from_entry = "Long", stop = longStopPx, comment_loss = "SL")
        else
            strategy.exit("Long-RUN", from_entry = "Long", stop = longStopPx, comment_loss = "SL")
        // [R16] trade lines
        if not na(lnLongEntry)
            line.delete(lnLongEntry)
        if not na(lnLongSL)
            line.delete(lnLongSL)
        if not na(lnLongTP1)
            line.delete(lnLongTP1)
        lnLongEntry := line.new(bar_index, longEntryPx, bar_index + 1, longEntryPx, color = color.blue, width = 1)
        lnLongSL    := line.new(bar_index, longStopPx, bar_index + 1, longStopPx, color = color.red, width = 1)
        lnLongTP1   := line.new(bar_index, longTp1Px, bar_index + 1, longTp1Px, color = color.lime, width = 1)
        if not na(lnLongHint)
            line.set_x2(lnLongHint, bar_index)
        if not na(lnLongImp)
            line.set_x2(lnLongImp, bar_index)
        if not na(bxLongFvg)
            box.set_right(bxLongFvg, bar_index)
    else if strategy.position_size == 0
        // Still flat — only then evaluate cancel / pending limit
        bool cancelLongEntry = false

        // [R7] no FVG after window → skip & count
        if not longFvgFound and bar_index > longBosBar + 3
            skippedNoFvgLong += 1
            cancelLongEntry := true
        // structure fail
        if close < longL2
            cancelLongEntry := true
        // entry timeout
        if bar_index - longBosBar >= entryTimeout
            cancelLongEntry := true

        if cancelLongEntry
            strategy.cancel("Long")
            if not na(lnLongHint)
                line.delete(lnLongHint)
                lnLongHint := na
            if not na(lnLongImp)
                line.delete(lnLongImp)
                lnLongImp := na
            if not na(bxLongFvg)
                box.delete(bxLongFvg)
                bxLongFvg := na
            longState    := S_IDLE
            longFvgFound := false
        else if longFvgFound
            float ePx = f_longEntry(longFvgTop, longFvgBot)
            longEntryPx := ePx
            longTp1Px   := longImpulseHigh
            longStopPx  := f_longStop(ePx, longL2, longFvgBot, atr)

            // [R16] FVG box
            if na(bxLongFvg)
                bxLongFvg := box.new(math.max(longFvgBar - 2, longL2Bar), longFvgTop, bar_index, longFvgBot, bgcolor = color.new(color.green, 85), border_color = color.new(color.green, 50))
            else
                box.set_right(bxLongFvg, bar_index)
                box.set_top(bxLongFvg, longFvgTop)
                box.set_bottom(bxLongFvg, longFvgBot)

            // [R7][F4] missed trade: close through prior impulse extreme without trading the limit
            if flat and bar_index > longBosBar and close > prevImp and low > ePx
                strategy.cancel("Long")
                if not na(lnLongHint)
                    line.delete(lnLongHint)
                    lnLongHint := na
                if not na(lnLongImp)
                    line.delete(lnLongImp)
                    lnLongImp := na
                if not na(bxLongFvg)
                    box.delete(bxLongFvg)
                    bxLongFvg := na
                longState := S_IDLE
            else if flat and inDateRange
                // [R7][R8] limit entry
                strategy.entry("Long", strategy.long, limit = ePx, comment = "Long FVG")

// [R10][R11] TRADE
if longState == S_TRADE
    if strategy.position_size > 0
        if na(longPosMax) or strategy.position_size > longPosMax
            longPosMax := strategy.position_size
        // [R10] TP1 filled via position size drop — reprice runner only [F3]
        if not longTp1Done and not na(longPosMax) and strategy.position_size < longPosMax
            longTp1Done := true
            if beAfterTp1
                longStopPx := strategy.position_avg_price
                strategy.exit("Long-RUN", from_entry = "Long", stop = longStopPx, comment_loss = "BE/SL")
        // [R16] extend trade lines
        if not na(lnLongEntry)
            line.set_x2(lnLongEntry, bar_index)
        if not na(lnLongSL)
            line.set_x2(lnLongSL, bar_index)
            line.set_y1(lnLongSL, longStopPx)
            line.set_y2(lnLongSL, longStopPx)
        if not na(lnLongTP1)
            line.set_x2(lnLongTP1, bar_index)

        // [R11] runner (hard stop remains via strategy.exit Long-RUN)
        bool runnerExitL = exitMode == "supertrend" ? stFlipBear : close < ma
        if runnerExitL
            strategy.close("Long", comment = exitMode == "supertrend" ? "ST runner" : "MA runner")
    else
        // flat — clean up
        strategy.cancel("Long")
        if not na(lnLongHint)
            line.delete(lnLongHint)
            lnLongHint := na
        if not na(lnLongImp)
            line.delete(lnLongImp)
            lnLongImp := na
        if not na(bxLongFvg)
            box.delete(bxLongFvg)
            bxLongFvg := na
        if not na(lnLongEntry)
            line.delete(lnLongEntry)
            lnLongEntry := na
        if not na(lnLongSL)
            line.delete(lnLongSL)
            lnLongSL := na
        if not na(lnLongTP1)
            line.delete(lnLongTP1)
            lnLongTP1 := na
        longState    := S_IDLE
        longFvgFound := false
        longTp1Done  := false
        longPosMax   := na

// =============================================================================
// SHORT state machine [R12] — full mirror
// =============================================================================
if enableShorts
    if (shortState == S_ARMED or shortState == S_ENTRY) and not na(lnShortLint) and not na(shortLint)
        line.set_x2(lnShortLint, bar_index)
        line.set_y1(lnShortLint, shortLint)
        line.set_y2(lnShortLint, shortLint)

    // ARMED
    if shortState == S_ARMED
        if close > shortH2
            strategy.cancel("Short")
            if not na(lnShortLint)
                line.delete(lnShortLint)
                lnShortLint := na
            shortState := S_IDLE
        else if bar_index - shortArmedBar >= armTimeout
            strategy.cancel("Short")
            if not na(lnShortLint)
                line.delete(lnShortLint)
                lnShortLint := na
            shortState := S_IDLE
        else if close < shortLint
            shortState       := S_ENTRY
            shortBosBar      := bar_index
            shortImpulseLow  := low
            markShortBos     := true
            // Bearish FVG scan H2 .. BoS+3
            [fndS, tS, bS, fbS] = f_scanBearFvg(shortH2Bar, shortBosBar + 3)
            shortFvgFound := fndS
            shortFvgTop   := tS
            shortFvgBot   := bS
            shortFvgBar   := fbS
            if not na(lnShortImp)
                line.delete(lnShortImp)
            lnShortImp := line.new(shortBosBar, shortImpulseLow, bar_index, shortImpulseLow, color = color.new(color.orange, 0), width = 1, style = line.style_dashed)

    // ENTRY
    if shortState == S_ENTRY
        // [F4] capture prior impulse extreme before this bar's update
        float prevImpLow = shortImpulseLow
        shortImpulseLow := math.min(nz(shortImpulseLow, low), low)
        if not na(lnShortImp)
            line.set_x2(lnShortImp, bar_index)
            line.set_y1(lnShortImp, shortImpulseLow)
            line.set_y2(lnShortImp, shortImpulseLow)

        if bar_index >= shortH2Bar and bar_index <= shortBosBar + 3
            if high < low[2]
                shortFvgTop   := low[2]
                shortFvgBot   := high
                shortFvgBar   := bar_index
                shortFvgFound := true

        // [F2] FIRST: fill check — never cancel while holding a filled short
        if strategy.position_size < 0 and strategy.position_size[1] >= 0
            shortState   := S_TRADE
            shortPosMax  := math.abs(strategy.position_size)
            shortTp1Done := false
            shortEntryPx := strategy.position_avg_price
            shortTp1Px   := shortImpulseLow
            shortStopPx  := f_shortStop(shortEntryPx, shortH2, shortFvgTop, atr)
            // [R9][R10][F3] TP1 tranche + runner always stop-protected
            if tp1Pct > 0
                strategy.exit("Short-TP1", from_entry = "Short", qty_percent = tp1Pct, limit = shortTp1Px, stop = shortStopPx, comment_profit = "TP1", comment_loss = "SL")
                strategy.exit("Short-RUN", from_entry = "Short", stop = shortStopPx, comment_loss = "SL")
            else
                strategy.exit("Short-RUN", from_entry = "Short", stop = shortStopPx, comment_loss = "SL")
            if not na(lnShortEntry)
                line.delete(lnShortEntry)
            if not na(lnShortSL)
                line.delete(lnShortSL)
            if not na(lnShortTP1)
                line.delete(lnShortTP1)
            lnShortEntry := line.new(bar_index, shortEntryPx, bar_index + 1, shortEntryPx, color = color.blue, width = 1)
            lnShortSL    := line.new(bar_index, shortStopPx, bar_index + 1, shortStopPx, color = color.red, width = 1)
            lnShortTP1   := line.new(bar_index, shortTp1Px, bar_index + 1, shortTp1Px, color = color.lime, width = 1)
            if not na(lnShortLint)
                line.set_x2(lnShortLint, bar_index)
            if not na(lnShortImp)
                line.set_x2(lnShortImp, bar_index)
            if not na(bxShortFvg)
                box.set_right(bxShortFvg, bar_index)
        else if strategy.position_size == 0
            // Still flat — only then evaluate cancel / pending limit
            bool cancelShortEntry = false
            if not shortFvgFound and bar_index > shortBosBar + 3
                skippedNoFvgShort += 1
                cancelShortEntry := true
            if close > shortH2
                cancelShortEntry := true
            if bar_index - shortBosBar >= entryTimeout
                cancelShortEntry := true

            if cancelShortEntry
                strategy.cancel("Short")
                if not na(lnShortLint)
                    line.delete(lnShortLint)
                    lnShortLint := na
                if not na(lnShortImp)
                    line.delete(lnShortImp)
                    lnShortImp := na
                if not na(bxShortFvg)
                    box.delete(bxShortFvg)
                    bxShortFvg := na
                shortState    := S_IDLE
                shortFvgFound := false
            else if shortFvgFound
                float ePxS = f_shortEntry(shortFvgTop, shortFvgBot)
                shortEntryPx := ePxS
                shortTp1Px   := shortImpulseLow
                shortStopPx  := f_shortStop(ePxS, shortH2, shortFvgTop, atr)

                if na(bxShortFvg)
                    bxShortFvg := box.new(math.max(shortFvgBar - 2, shortH2Bar), shortFvgTop, bar_index, shortFvgBot, bgcolor = color.new(color.red, 85), border_color = color.new(color.red, 50))
                else
                    box.set_right(bxShortFvg, bar_index)
                    box.set_top(bxShortFvg, shortFvgTop)
                    box.set_bottom(bxShortFvg, shortFvgBot)

                // [R7][R12][F4] missed: close through prior impulse low without trading the limit
                if flat and bar_index > shortBosBar and close < prevImpLow and high < ePxS
                    strategy.cancel("Short")
                    if not na(lnShortLint)
                        line.delete(lnShortLint)
                        lnShortLint := na
                    if not na(lnShortImp)
                        line.delete(lnShortImp)
                        lnShortImp := na
                    if not na(bxShortFvg)
                        box.delete(bxShortFvg)
                        bxShortFvg := na
                    shortState := S_IDLE
                else if flat and inDateRange
                    strategy.entry("Short", strategy.short, limit = ePxS, comment = "Short FVG")

    // TRADE
    if shortState == S_TRADE
        if strategy.position_size < 0
            float absSz = math.abs(strategy.position_size)
            if na(shortPosMax) or absSz > shortPosMax
                shortPosMax := absSz
            // [R10] TP1 filled — reprice runner only [F3]
            if not shortTp1Done and not na(shortPosMax) and absSz < shortPosMax
                shortTp1Done := true
                if beAfterTp1
                    shortStopPx := strategy.position_avg_price
                    strategy.exit("Short-RUN", from_entry = "Short", stop = shortStopPx, comment_loss = "BE/SL")
            if not na(lnShortEntry)
                line.set_x2(lnShortEntry, bar_index)
            if not na(lnShortSL)
                line.set_x2(lnShortSL, bar_index)
                line.set_y1(lnShortSL, shortStopPx)
                line.set_y2(lnShortSL, shortStopPx)
            if not na(lnShortTP1)
                line.set_x2(lnShortTP1, bar_index)

            bool runnerExitS = exitMode == "supertrend" ? stFlipBull : close > ma
            if runnerExitS
                strategy.close("Short", comment = exitMode == "supertrend" ? "ST runner" : "MA runner")
        else
            strategy.cancel("Short")
            if not na(lnShortLint)
                line.delete(lnShortLint)
                lnShortLint := na
            if not na(lnShortImp)
                line.delete(lnShortImp)
                lnShortImp := na
            if not na(bxShortFvg)
                box.delete(bxShortFvg)
                bxShortFvg := na
            if not na(lnShortEntry)
                line.delete(lnShortEntry)
                lnShortEntry := na
            if not na(lnShortSL)
                line.delete(lnShortSL)
                lnShortSL := na
            if not na(lnShortTP1)
                line.delete(lnShortTP1)
                lnShortTP1 := na
            shortState    := S_IDLE
            shortFvgFound := false
            shortTp1Done  := false
            shortPosMax   := na

// =============================================================================

plotshape(markLongL2, title = "Qualified L2", style = shape.triangleup, location = location.belowbar, color = color.green, size = size.tiny, text = "L2")
plotshape(markLongBos, title = "BoS Long", style = shape.diamond, location = location.abovebar, color = color.lime, size = size.tiny, text = "BoS")
